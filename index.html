Abstract

บทคัดย่อ

GUESSNUM เป็นเกมทายเลข 5 หลักที่ผู้เล่นใช้ Keypad 4x4 กดตัวเลข บอร์ด Arduino UNO R4 WiFi ฝั่ง Keypad ส่งเลขผ่านโปรโตคอล MQTT ไปยังบอร์ดฝั่งรับ ซึ่งสุ่มเฉลยไว้และตรวจคำตอบทีละหลัก ผลลัพธ์แสดงด้วยไฟ RGB 5 ดวง ได้แก่ สีเขียว (ถูกเลขและถูกตำแหน่ง) สีเหลือง (มีเลขนี้แต่ผิดตำแหน่ง) และสีแดง (ไม่มีเลขนี้) จอ LCD แสดงสถานะเกมและข้อความตอนชนะหรือแพ้ ส่วน 7-segment แสดงจำนวนครั้งที่ทายได้ที่เหลือ เมื่อจบเกม ระบบส่งผลการเล่นในรูปแบบ JSON ไปยังเว็บ Dashboard บน GitHub Pages เพื่อเก็บสถิติ เช่น จำนวนเกม อัตราการชนะ และประวัติการทายแต่ละครั้ง

GUESSNUM is a physical, Wordle-style 5-digit number guessing game. Players enter guesses on a 4x4 keypad connected to an Arduino UNO R4 WiFi, which sends them over MQTT to a second board. The receiver board holds a random answer and checks each digit. Five RGB LEDs show the result: green for the right digit in the right place, yellow for the right digit in the wrong place, and red for a digit not in the answer. An I2C LCD shows the game status, and a 4-digit 7-segment display counts the remaining tries. When a game ends, the result is published as JSON to a web dashboard on GitHub Pages that records the win rate and full guess history.

How to play

วิธีเล่น

กติกา

  1. ระบบสุ่มเลขเฉลย 5 หลัก (เลขซ้ำได้)
  2. กดเลข 5 หลักบน Keypad แล้วกด # เพื่อส่ง
  3. ไฟ RGB 5 ดวงบอกผลทีละหลัก
  4. ทายได้ 4 ครั้ง 7-segment นับถอยหลัง 4 → 3 → 2 → 1
  5. จบเกม LCD โชว์เฉลยค้าง 8 วินาที แล้วเริ่มเกมใหม่อัตโนมัติ

Technical highlights

จุดเด่นทางเทคนิค

🟩

ตรวจคำตอบแบบ Wordle

ให้สีเขียวก่อน แล้วค่อยให้สีเหลืองตามจำนวนเลขที่เหลือในเฉลย ทำให้กรณีเลขซ้ำได้สีที่ถูกต้อง

⌨️

สแกน Keypad เอง

UNO R4 ทำงานเร็ว (48 MHz) ไลบรารี Keypad.h จึงอ่านค่าก่อนสายนิ่งและได้เลขมั่ว เราเขียนตัวสแกนเอง โดยรอสัญญาณนิ่ง 20 µs, debounce 30 ms, ไม่รันซ้ำตอนกดค้าง และตัดทิ้งเมื่อกดหลายปุ่มพร้อมกัน

⏱️

ไม่ใช้ delay() ตอนกะพริบ

ใช้ millis() คุมจังหวะไฟกะพริบและการสลับหน้าจอ LCD บอร์ดจึงยังรับข้อความ MQTT ได้ตลอดเวลา

🔁

กันข้อมูลซ้ำบนเว็บ

ผลเกมส่งแบบ retained พร้อม uid เปิดเว็บทีหลังก็ยังได้ผลล่าสุด และเว็บไม่นับเกมเดิมซ้ำ

/*
  ===========================================================
   GUESSNUM – บอร์ดฝั่งรับ (Arduino UNO R4 WiFi)
  -----------------------------------------------------------
   - รับเลขทาย 5 หลักจาก MQTT (ส่งมาจากบอร์ด keypad)
   - สุ่มเฉลย 5 หลัก แล้วเช็คทีละตำแหน่ง -> แสดงสีบน RGB 5 ดวง
       เขียว = ถูกเลข ถูกตำแหน่ง
       เหลือง = มีเลขนี้ในเฉลย แต่ผิดตำแหน่ง
       แดง   = ไม่มีเลขนี้ในเฉลย
   - LCD I2C 16x2 : สถานะเกม + หน้าจอชนะ/แพ้แบบน่ารัก
   - 7-segment TM1637 : จำนวนครั้งที่เหลือ 4 -> 0
   - LCD โชว์เลขที่กำลังพิมพ์จาก keypad (topic /typing)
   - จบเกมแล้วส่งผลเป็น JSON ไปที่ MQTT ให้เว็บเก็บสถิติ
  -----------------------------------------------------------
   ไลบรารีที่ต้องลง (Sketch > Include Library > Manage Libraries)
   - ArduinoMqttClient   (by Arduino)
   - LiquidCrystal I2C   (by Frank de Brabander)
   - TM1637              (by Avishay Orpaz)
   * WiFiS3 มากับบอร์ด UNO R4 WiFi อยู่แล้ว
  ===========================================================
*/

#define COMMON_ANODE true   // ถ้า RGB เป็นแบบขาร่วมไฟบวก (common anode) เปลี่ยนเป็น true

const uint8_t LED_PIN[5][3] = {   // {R, G, B}
  {11, 12, 13},   // หลัก 1
  { 8,  9, 10},   // หลัก 2
  { 5,  6,  7},   // หลัก 3
  { 2,  3,  4},   // หลัก 4

#define SEG_CLK 1
#define SEG_DIO 0
TM1637Display seg(SEG_CLK, SEG_DIO);

LiquidCrystal_I2C lcd(0x27, 16, 2);   // ถ้าจอไม่ขึ้นตัวหนังสือ ลองเปลี่ยนเป็น 0x3F

// ================= กติกา / เวลา =================
const uint8_t MAX_TRIES = 4;                // ทายได้ 4 ครั้ง -> 7seg โชว์ 4,3,2,1 แพ้แล้วเป็น 0
const unsigned long REVEAL_MS     = 2000;   // โชว์สีของตาสุดท้ายค้างไว้ก่อนเริ่มกะพริบ
const unsigned long BLINK_MS      = 600;    // จังหวะกะพริบ (ช้า ๆ สบายตา)
const unsigned long FRAME_MS      = 2400;   // สลับหน้าจอ LCD ตอนจบเกม
const unsigned long END_SCREEN_MS = 8000;   // โชว์เฉลยค้าง 8 วิ แล้วเริ่มเกมใหม่อัตโนมัติ

// ================= ตัวอักษรพิเศษบน LCD =================
byte chHeart[8] = {0b00000,0b01010,0b11111,0b11111,0b01110,0b00100,0b00000,0b00000};
byte chSmile[8] = {0b00000,0b01010,0b01010,0b00000,0b10001,0b01110,0b00000,0b00000};
byte chSad[8]   = {0b00000,0b01010,0b01010,0b00000,0b01110,0b10001,0b00000,0b00000};
byte chStar[8]  = {0b00100,0b10101,0b01110,0b11111,0b01110,0b10101,0b00100,0b00000};
byte chNote[8]  = {0b00010,0b00011,0b00010,0b00010,0b01110,0b11110,0b01100,0b00000};
byte chTear[8]  = {0b00100,0b00100,0b01010,0b01010,0b10001,0b10001,0b01110,0b00000};

#define HEART "\x01"
#define SMILE "\x02"
#define SAD   "\x03"
#define STAR  "\x04"
#define NOTE  "\x05"
#define TEAR  "\x06"

// ================= คำบน 7-segment =================
const uint8_t SEG_GOOD[] = {
  SEG_A | SEG_C | SEG_D | SEG_E | SEG_F,          // G
  SEG_C | SEG_D | SEG_E | SEG_G,                  // o
  SEG_C | SEG_D | SEG_E | SEG_G,                  // o
  SEG_B | SEG_C | SEG_D | SEG_E | SEG_G           // d
};
const uint8_t SEG_LOSE[] = {
  SEG_D | SEG_E | SEG_F,                          // L
  SEG_A | SEG_B | SEG_C | SEG_D | SEG_E | SEG_F,  // O
  SEG_A | SEG_C | SEG_D | SEG_F | SEG_G,          // S
  SEG_A | SEG_D | SEG_E | SEG_F | SEG_G           // E
};

// ================= สถานะเกม =================
enum LedColor : uint8_t { C_OFF, C_RED, C_YELLOW, C_GREEN };
const char *COLOR_NAME[] = { "off", "red", "yellow", "green" };

enum GameState { PLAYING, WON, LOST };
GameState state = PLAYING;

char     answer[6];
char     gameUid[9];
uint8_t  triesUsed = 0;
char     guessLog[MAX_TRIES][6];
LedColor colorLog[MAX_TRIES][5];

unsigned long endAt = 0, lastBlink = 0;
bool   blinkOn  = false;
int8_t lcdFrame = -1;

WiFiClient wifiClient;
MqttClient mqttClient(wifiClient);

// =====================================================
//                      LED / จอ
// =====================================================
void writeLedPin(uint8_t pin, bool on) {
  digitalWrite(pin, (on != COMMON_ANODE) ? HIGH : LOW);
}

void setLed(uint8_t i, LedColor c) {
  writeLedPin(LED_PIN[i][0], c == C_RED   || c == C_YELLOW);  // R
  writeLedPin(LED_PIN[i][1], c == C_GREEN || c == C_YELLOW);  // G
  writeLedPin(LED_PIN[i][2], false);                          // B (ไม่ใช้)
}

void allLeds(LedColor c) {
  for (uint8_t i = 0; i < 5; i++) setLed(i, c);
}

// พิมพ์ 1 บรรทัดเต็ม 16 ช่อง (เติมช่องว่างลบของเก่า)
void lcdLine(uint8_t row, const char *text) {
  lcd.setCursor(0, row);
  uint8_t n = 0;
  while (text[n] && n < 16) { lcd.write((uint8_t)text[n]); n++; }
  while (n < 16) { lcd.write(' '); n++; }
}

// =====================================================
//                      ตัวเกม
// =====================================================
void newGame() {
  for (uint8_t i = 0; i < 5; i++) answer[i] = '0' + random(10);
  answer[5] = '\0';
  snprintf(gameUid, sizeof(gameUid), "%08lX", (unsigned long)random(0x7FFFFFFF));

  triesUsed = 0;
  state = PLAYING;
  allLeds(C_OFF);
  seg.showNumberDec(MAX_TRIES);

  lcdLine(0, " " STAR " GUESSNUM " STAR " " NOTE);
  lcdLine(1, " Guess 5 digit " SMILE);

  Serial.print(F("\n[NEW GAME] answer = "));   // ไว้เทส ตอนพรีเซนต์จริงลบออกได้
  Serial.println(answer);
}

// เช็คแบบ Wordle: เขียวก่อน แล้วค่อยแจกเหลืองตามจำนวนเลขที่เหลือในเฉลย
void checkGuess(const char *g, LedColor res[5]) {
  uint8_t remain[10] = {0};
  for (uint8_t i = 0; i < 5; i++) {
    if (g[i] == answer[i]) res[i] = C_GREEN;
    else { res[i] = C_RED; remain[answer[i] - '0']++; }
  }
  for (uint8_t i = 0; i < 5; i++) {
    if (res[i] == C_GREEN) continue;
    uint8_t d = g[i] - '0';
    if (remain[d] > 0) { res[i] = C_YELLOW; remain[d]--; }
  }
}

void handleGuess(const char *g) {
  LedColor res[5];
  checkGuess(g, res);

  strcpy(guessLog[triesUsed], g);
  memcpy(colorLog[triesUsed], res, sizeof(res));
  triesUsed++;

  bool win = true;
  for (uint8_t i = 0; i < 5; i++) {
    setLed(i, res[i]);
    if (res[i] != C_GREEN) win = false;
  }

  char line[17];
  snprintf(line, sizeof(line), "Try %u:  %s", triesUsed, g);   // เลขเริ่ม column 9 ตรงกับ Result
  lcdLine(0, line);
  char pat[17] = "Result: ";
  for (uint8_t i = 0; i < 5; i++) pat[8 + i] = "-RYG"[res[i]];
  pat[13] = '\0';
  lcdLine(1, pat);

  Serial.print(F("Guess ")); Serial.print(g);
  Serial.print(F(" -> ")); Serial.println(pat + 8);

  if (win)                          endGame(true);
  else if (triesUsed >= MAX_TRIES)  endGame(false);
  else                              seg.showNumberDec(MAX_TRIES - triesUsed);
}

void endGame(bool win) {
  state     = win ? WON : LOST;
  endAt     = millis();
  lastBlink = endAt;
  blinkOn   = false;
  lcdFrame  = -1;
  seg.setSegments(win ? SEG_GOOD : SEG_LOSE);
  publishGame(win);
  Serial.println(win ? F("*** WIN ***") : F("*** LOSE ***"));
}

// หน้าจอ LCD ตอนจบเกม: บรรทัดบนสลับข้อความ, บรรทัดล่างโชว์เฉลยค้างตลอด
void drawEndFrame(int8_t f) {
  if (state == WON) {
    if (f == 0) lcdLine(0, " " HEART " YOU WIN!! " HEART);
    else        lcdLine(0, STAR " YAY! (^o^)/ " STAR);
  } else {
    if (f == 0) lcdLine(0, " " SAD " GAME OVER " SAD);
    else        lcdLine(0, " (T_T) " TEAR " oh no");
  }
  char line[17];
  snprintf(line, sizeof(line), "Answer: %s %s", answer, state == WON ? SMILE : TEAR);
  lcdLine(1, line);   // เลขเฉลยเริ่ม column 9 เหมือน Try / Result
}

// เรียกทุกรอบ loop: กะพริบไฟ + สลับหน้าจอ แบบไม่ใช้ delay
void updateEndScreen() {
  if (state == PLAYING) return;
  unsigned long now = millis();
  unsigned long t   = now - endAt;

  int8_t f = (t / FRAME_MS) % 2;
  if (f != lcdFrame) { lcdFrame = f; drawEndFrame(f); }

  if (t >= REVEAL_MS && now - lastBlink >= BLINK_MS) {
    lastBlink = now;
    blinkOn = !blinkOn;
    if (state == WON) {
      allLeds(blinkOn ? C_GREEN : C_OFF);
      if (blinkOn) seg.setSegments(SEG_GOOD); else seg.clear();
    } else {
      allLeds(blinkOn ? C_RED : C_OFF);
      if (blinkOn) seg.setSegments(SEG_LOSE); else seg.showNumberDec(0);
    }
  }

  if (t >= END_SCREEN_MS) newGame();
}

// =====================================================
//                  ส่งสถิติไปเว็บ (JSON)
// =====================================================
void publishGame(bool win) {
  String js;
  js.reserve(700);
  js += F("{\"uid\":\"");        js += gameUid;
  js += F("\",\"result\":\"");   js += win ? "WIN" : "LOSE";
  js += F("\",\"attempts\":");   js += triesUsed;
  js += F(",\"maxAttempts\":");  js += MAX_TRIES;
  js += F(",\"answer\":\"");     js += answer;
  js += F("\",\"guesses\":[");
  for (uint8_t t = 0; t < triesUsed; t++) {
    if (t) js += ',';
    js += F("{\"number\":\"");   js += guessLog[t];
    js += F("\",\"colors\":[");
    for (uint8_t i = 0; i < 5; i++) {
      if (i) js += ',';
      js += '"'; js += COLOR_NAME[colorLog[t][i]]; js += '"';
    }
    js += F("]}");
  }
  js += F("]}");

  if (mqttClient.connected()) {
    // retain = true -> เปิดเว็บทีหลังก็ยังได้ผลเกมล่าสุด (เว็บกันซ้ำด้วย uid)
    mqttClient.beginMessage(TOPIC_GAME, (unsigned long)js.length(), true, 1);
    mqttClient.print(js);
    mqttClient.endMessage();
    Serial.print(F("Published: ")); Serial.println(js);
  } else {
    Serial.println(F("MQTT not connected - stats not sent"));
  }
}

bool isFiveDigits(const char *s) {
  if (strlen(s) != 5) return false;
  for (uint8_t i = 0; i < 5; i++) if (!isdigit((unsigned char)s[i])) return false;
  return true;
}

void onMqttMessage(int messageSize) {
  String topic = mqttClient.messageTopic();

  char msg[24];
  int n = 0;
  while (mqttClient.available()) {
    char c = (char)mqttClient.read();
    if (n < (int)sizeof(msg) - 1) msg[n++] = c;
  }
  msg[n] = '\0';

  // ตัดช่องว่าง / ขึ้นบรรทัด หัว-ท้าย
  char *p = msg;
  while (*p && isspace((unsigned char)*p)) p++;
  int len = strlen(p);
  while (len > 0 && isspace((unsigned char)p[len - 1])) p[--len] = '\0';

  Serial.print(F("MQTT [")); Serial.print(topic); Serial.print(F("] ")); Serial.println(p);

  if (topic == TOPIC_TYPING) {          // โชว์เลขที่กำลังพิมพ์
    if (state == PLAYING) {
      char line[17];
      snprintf(line, sizeof(line), " Input: %s", p);
      lcdLine(1, line);
    }
    return;
  }
  if (topic != TOPIC_GUESS) return;

  // คำสั่งเริ่มเกมใหม่: "NEW" หรือ "*"
  if (!strcmp(p, "NEW") || !strcmp(p, "new") || !strcmp(p, "*")) { newGame(); return; }

  if (state != PLAYING) {
    Serial.println(F("Game ended - waiting for new game"));
    return;
  }

  if (!isFiveDigits(p)) {
    lcdLine(0, " Oops! " SAD " 5 digit");
    char line[17];
    snprintf(line, sizeof(line), " got: %s", p);
    lcdLine(1, line);
    return;
  }

  handleGuess(p);
}

void keypadBegin() {
  for (uint8_t r = 0; r < ROWS; r++) pinMode(rowPins[r], INPUT_PULLUP);
  for (uint8_t c = 0; c < COLS; c++) pinMode(colPins[c], INPUT);   // column ที่ไม่ได้ใช้ = ลอย
}

// สแกน 1 รอบ: คืนค่าปุ่มที่กด, KEY_NONE ถ้าไม่กด, KEY_MULTI ถ้ากดหลายปุ่ม
char scanRaw() {
  char found = KEY_NONE;
  uint8_t count = 0;

  for (uint8_t c = 0; c < COLS; c++) {
    // เปิด column นี้ด้วย LOW
    pinMode(colPins[c], OUTPUT);
    digitalWrite(colPins[c], LOW);
    delayMicroseconds(SETTLE_US);            // รอสายนิ่งก่อนอ่าน

    for (uint8_t r = 0; r < ROWS; r++) {
      if (digitalRead(rowPins[r]) == LOW) {
        found = keys[r][c];
        count++;
      }
    }

    // ปิด column: ดัน HIGH สั้น ๆ ให้สาย row ชาร์จกลับเร็ว แล้วปล่อยลอย
    digitalWrite(colPins[c], HIGH);
    delayMicroseconds(2);
    pinMode(colPins[c], INPUT);
    delayMicroseconds(SETTLE_US);            // รอก่อนไป column ถัดไป
  }

  if (count > 1) return KEY_MULTI;           // กัน ghost key
  return found;
}

// คืนค่าปุ่ม "เฉพาะตอนเริ่มกด" หลัง debounce แล้ว (กดค้างไม่รันซ้ำ)
char getKey() {
  static char lastRaw = KEY_NONE;
  static char stable  = KEY_NONE;
  static unsigned long changedAt = 0;

  char raw = scanRaw();
  if (raw != lastRaw) {                      // ค่าเปลี่ยน -> เริ่มจับเวลาใหม่
    lastRaw = raw;
    changedAt = millis();
    return KEY_NONE;
  }

  if (millis() - changedAt >= DEBOUNCE_MS && raw != stable) {
    stable = raw;                            // นิ่งครบ 30 ms แล้ว
    if (stable != KEY_NONE && stable != KEY_MULTI) return stable;
  }
  return KEY_NONE;
}

void handleKey(char k) {
  Serial.print(F("Key: ")); Serial.println(k);

  if (k >= '0' && k <= '9') {                 // พิมพ์เลข
    if (len < 5) {
      input[len++] = k;
      input[len] = '\0';
      showInput();
      sendTyping();
    } else {
      flashIcon(ICON_CROSS, 300);             // ครบ 5 หลักแล้ว
    }
  }
  else if (k == '*') {                        // ลบตัวท้าย
    if (len > 0) input[--len] = '\0';
    showInput();
    sendTyping();
  }
  else if (k == 'C') {                        // ล้างทั้งหมด
    clearInput();
    showInput();
    sendTyping();
  }
  else if (k == '#') {                        // ส่งคำตอบ
    if (len == 5 && publishText(TOPIC_GUESS, input, 1)) {
      Serial.print(F("Sent guess: ")); Serial.println(input);
      clearInput();
      flashIcon(ICON_CHECK, 600);
      // ไม่ส่ง typing ตรงนี้ เพื่อให้ LCD ฝั่งรับโชว์ผลการทายค้างไว้
    } else {
      flashIcon(ICON_CROSS, 500);             // ยังไม่ครบ 5 หลัก / ส่งไม่ได้
    }
  }
  else if (k == 'A') {                        // เริ่มเกมใหม่
    if (publishText(TOPIC_GUESS, "NEW", 1)) {
      Serial.println(F("Sent NEW"));
      clearInput();
      flashIcon(ICON_HEART, 800);
    } else {
      flashIcon(ICON_CROSS, 500);
    }
  }
  // B, D ไม่ได้ใช้
}

// ------------------------------

const MQTT_URL   = "wss://mqtt-dashboard.com:8884/mqtt";
const TOPIC_GAME = "r2n/game";

const STORAGE_KEY = "numdle_games";
const SEEN_KEY    = "numdle_seen";


// ------------------------------
// Load / Save (เก็บประวัติไว้ในเบราว์เซอร์)
// ------------------------------

function loadJSON(key) {
  try { return JSON.parse(localStorage.getItem(key)) || []; }
  catch (e) { return []; }
}

function saveJSON(key, value) {
  try { localStorage.setItem(key, JSON.stringify(value)); }
  catch (e) {}
}

let games = loadJSON(STORAGE_KEY);
let seen  = loadJSON(SEEN_KEY);


// ------------------------------
// Render Latest Game
// ------------------------------

function renderLatest() {

  const g = games[0];
  const resultEl = document.getElementById("latestResult");

  if (!g) {
    resultEl.textContent = "-";
    document.getElementById("latestAttempts").textContent = "-";
    document.getElementById("latestAnswer").textContent = "-----";
    return;
  }

  resultEl.textContent = g.result;
  resultEl.style.color = g.result === "WIN" ? "#2e9d58" : "#d84a4a";

  document.getElementById("latestAttempts").textContent =
    `${g.attempts} / ${g.maxAttempts || 5}`;

  document.getElementById("latestAnswer").textContent = g.answer;

}


// ------------------------------
// Render History
// ------------------------------

const table = document.getElementById("historyTable");

function renderHistory() {

  table.innerHTML = "";

  games.forEach(game => {

    const row = document.createElement("tr");

    row.innerHTML = `
      <td>#${game.id}</td>

      <td class="${game.result === "WIN" ? "win" : "lose"}">
        ${game.result}
      </td>

      <td>${game.attempts} / ${game.maxAttempts || 5}</td>

      <td>${game.answer}</td>

      <td>
        <button onclick="showGame(${game.id})">
          View
        </button>
      </td>
    `;

    table.appendChild(row);

  });

}

// Show Game Detail

function showGame(id) {

  const game = games.find(g => g.id === id);

  document.getElementById("gameDetail").style.display = "block";

  document.getElementById("detailAnswer").textContent =
    game.answer;

  const container =
    document.getElementById("guessHistory");

  container.innerHTML = "";

  game.guesses.forEach((guess, index) => {

    const row = document.createElement("div");

    row.className = "guess-row";

    row.appendChild(label);

    for(let i = 0; i < 5; i++) {

      const box = document.createElement("div");

      box.className =

      box.textContent =
        guess.number[i];

      row.appendChild(box);

    }

    container.appendChild(row);

  });

}

// Calculate Statistics
// ------------------------------

function updateStats(){

  const total = games.length;

  const wins =
    games.filter(g => g.result === "WIN").length;

  const losses =
    total - wins;

  const rate = total
    ? ((wins / total) * 100).toFixed(1)
    : "0";

  document.getElementById("totalGames")
    .textContent = total;

  document.getElementById("wins")
    .textContent = wins;

  document.getElementById("losses")
    .textContent = losses;

  document.getElementById("winRate")
    .textContent = rate + "%";

}

function renderAll() {
  renderLatest();
  renderHistory();
  updateStats();
}


// ------------------------------
// Add Game from Arduino
// ------------------------------

function addGame(data) {

  if (!data || !data.answer) return;

  // กันข้อมูลซ้ำ (retained message จะถูกส่งมาใหม่ทุกครั้งที่เปิดเว็บ)
  if (data.uid && seen.includes(data.uid)) return;

  const nextId = games.length
    ? Math.max(...games.map(g => g.id)) + 1
    : 1;

  games.unshift({
    id: nextId,
    uid: data.uid,
    result: data.result,
    attempts: data.attempts,
    maxAttempts: data.maxAttempts || 5,
    answer: data.answer,
    guesses: data.guesses || []
  });

  if (data.uid) {
    seen.push(data.uid);
    seen = seen.slice(-300);
    saveJSON(SEEN_KEY, seen);
  }

  saveJSON(STORAGE_KEY, games);
  renderAll();

}

function clearHistory() {

  if (!confirm("ลบประวัติการเล่นทั้งหมด?")) return;

  games = [];
  saveJSON(STORAGE_KEY, games);
  document.getElementById("gameDetail").style.display = "none";
  renderAll();

}

  statusEl.style.color = color;
}

const client = mqtt.connect(MQTT_URL, {
  clientId: "numdle-web-" + Math.random().toString(16).slice(2, 8)
});

client.on("connect", () => {
  setStatus("online", "#2e9d58");
  client.subscribe(TOPIC_GAME, { qos: 1 });
});

client.on("reconnect", () => setStatus("reconnecting...", "#d6ad21"));
client.on("offline",   () => setStatus("offline", "#d84a4a"));
client.on("error", err => console.error("MQTT error:", err));

client.on("message", (topic, payload) => {
  try {
    addGame(JSON.parse(payload.toString()));
  } catch (e) {
    console.warn("Bad message:", payload.toString());
  }
});
